// argv: codex exec --json --sandbox <mode> -c approval_policy="never"
//       --skip-git-repo-check -C <cwd> [resume <threadId>] -

type codexTranslator struct { pendingAgentMessage string /* ... */ }
// thread.started                          → Event{system, SessionID: threadId}
// item.started/completed(command_execution)→ tool_use / tool_result
// item.completed(agent_message)           → 先进 pendingAgentMessage 缓存：
//     下一条 agent_message 到来时，前一条作为 Event{text} 发出；
//     turn.completed 时，缓存的最后一条作为 final_text 随 Event{done} 发出
为什么缓存 agent_message：codex 把中间说明和最终答复都发成 agent_message，无法当场区分。延迟一拍——前一条一定是中间文本，最后一条才是 final_text。

func (s *ocServer) eventLoop(directory string)
// GET /event?directory=<dir> SSE 长连（实测：?directory=X 的会话事件
// 只出现在对应工作区流，不在全局总线）；断线指数退避重连（1s → 15s 上限）；
// 每个事件按 sessionID 分发到对应 run channel

func translate(ocEvent) []Event
// partKinds 表区分 text / reasoning delta → Event{text/thinking}
// message.part.updated：工具状态机 pending → running → completed
// session.idle → Event{done}；session.error(MessageAbortedError) → 中断收尾

// run 路径：sessionFor → POST /session/{id}/prompt_async?directory=<cwd>（异步）→
// 立刻 EventSystem{SessionID}（供 bridge 写 sessions/bindings）→ 目录级 SSE 回流事件

// 健壮性双保险：
// frameWatchdog(dir)：75s 无帧（含 heartbeat）→ cancel 强制重连
// pollRun(run)：每 3s GET /session/{id}/message 对账，reconcilePoll 补发丢失文本后再发终态
// ocRun.Stop() = POST /session/{id}/abort
// ListSessions(limit)：GET /session，按 time.updated 排序（/sessions 命令）
选 serve 而非 ACP 的原因（与实现直接相关）：单进程多 session 是一等公民（sessions 映射 + 按工作目录的 SSE 流按 sessionID 扇出）、有 /global/health 做存活探测、事件粒度更细（part 级 delta 与工具全生命周期）。会话由 opencode 磁盘持久化，bridge 重启后凭 sessionID 续聊。

type PendingQueue struct { /* 每 scope 一个；600ms 防抖窗口 */ }
